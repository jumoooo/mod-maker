import { expect, mock, test } from 'claude-code/testing'

const SURFACES = ['terminal', 'desktop'] as const

const band = (bodyColumns: number) => ({
  component: 'AbovePrompt' as const,
  requestId: 'band',
  props: { hasSurvey: false, isWorking: true, maxRows: 10, bodyColumns, scroll: { top: 0, bodyRows: 9 } as never, view: {} as never },
})

test('넓은 띠 — 요청 원문·단계줄·상태 낱말(당신 차례·진행 중·끝남)이 두 표면에 다 보인다', async ($, on) => {
  mock.clock(on)
  for (const surface of SURFACES) {
    const ui = await $.ui.mount({ plugin: 'sample-band', surface, ...band(100) })
    expect(await ui.find({ text: /주간 보고서 마무리/ })).toBeDefined()
    expect(await ui.find({ text: /2\/3 단계/ })).toBeDefined()
    expect(await ui.find({ text: /당신 차례/ })).toBeDefined()
    expect(await ui.find({ text: /진행 중/ })).toBeDefined()
    expect(await ui.find({ text: /끝남/ })).toBeDefined()
    await ui.unmount()
  }
})

test('아주 좁은 띠 — 한 줄 요약으로 내려가고 기다리는 사람 수가 먼저 온다', async ($, on) => {
  mock.clock(on)
  const ui = await $.ui.mount({ plugin: 'sample-band', surface: 'terminal', ...band(40) })
  expect(await ui.find({ text: /할 일 3/ })).toBeDefined()
  expect(await ui.find({ text: /당신 차례 1/ })).toBeDefined()
  expect(await ui.find({ text: /1\/3 끝남/ })).toBeDefined()
  await ui.unmount()
})

test('비었을 때 — 빈 상태 두 줄을 그린다', async ($, on) => {
  mock.clock(on)
  await $.command.run({ command: 'sample-band', args: 'clear', origin: {} as never, presentation: { isFullscreen: false, columns: 100 } })
  const ui = await $.ui.mount({ plugin: 'sample-band', surface: 'terminal', ...band(100) })
  expect(await ui.find({ text: /할 일이 비어 있습니다/ })).toBeDefined()
  expect(await ui.find({ text: /sample-band demo/ })).toBeDefined()
  await ui.unmount()
})
